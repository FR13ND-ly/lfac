class Robot {
    let id : int;
    function init(i : int) : void {
        id = i;
    }
    function check(v : int) : bool {
        if (id > v) {
            return true;
        }
        return false;
    }
}

function compute(a : int) : int {
    return a * 10;
}

let r : Robot;
let x : int;
let b : bool;

main {
    r = new Robot();
    r.init(50);
    
    x = compute(5);
    b = r.check(x) && (x == 50);

    if (b == true) {
        Print("OK");
    }

    while (x > 48) {
        Print(x);
        x = x - 1;
    }
}
